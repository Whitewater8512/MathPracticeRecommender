$$
= \int_{0}^{\pi} \sin \theta \cos \theta d\theta \left. \frac{r^4}{4} \right|_0^{\sqrt{2}} + 2 \int_{0}^{\pi} \sin \theta \cos \theta d\theta \left. \frac{r^4}{4} \right|_0^{\sqrt[4]{2}}
$$

$$
= \frac{1}{4} \int_{0}^{\pi} \sin \theta \cos \theta d\theta + 2 \times \frac{1}{4} \int_{0}^{\pi} \sin \theta \cos \theta d\theta \quad (\text{凑微分})
$$

$$
= \frac{1}{4} \times \frac{1}{2} \sin^2 \theta \left|_{0}^{\pi} \right. + \frac{1}{2} \times \frac{1}{2} \sin^2 \theta \left|_{0}^{\pi} \right. = \frac{1}{8} + \frac{1}{4} = \frac{3}{8}.
$$
